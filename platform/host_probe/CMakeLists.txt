cmake_minimum_required(VERSION 3.20)
project(DK64RuntimeProbe LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(generated "${repo}/build/RecompiledFuncsVita")
if(NOT EXISTS "${generated}/recomp_overlays.inl")
    message(FATAL_ERROR "Run N64Recomp us.vita.toml first")
endif()
option(PROBE_ASAN "Instrument the native probe for memory errors" ON)
option(PROBE_GL "Exercise the shared GL draw path with offscreen Mesa" OFF)
if(PROBE_ASAN)
    add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address)
endif()
# Use the Vita address aliases and memory budget, without pretending to be the
# Vita SDK. Host OS threads and a capture-only sink isolate the runtime.
add_compile_definitions(RECOMP_MIRROR_KSEG1 RECOMP_RDRAM_SIZE=33554432)
add_compile_options(-fno-strict-aliasing -ffunction-sections -fdata-sections)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
    add_compile_options(-msse4.1)
endif()
set(RECOMP_ENABLE_MODS OFF CACHE BOOL "" FORCE)
set(RT64_FAST ON CACHE BOOL "" FORCE)
set(RT64_FAST_VITAGL OFF CACHE BOOL "" FORCE)
set(RT64_FAST_GLES2 ${PROBE_GL} CACHE BOOL "" FORCE)
set(RT64_FAST_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${repo}/lib/N64ModernRuntime" runtime)
add_subdirectory("${repo}/lib/rt64" rt64)
target_include_directories(ultramodern PUBLIC "${repo}/lib/N64ModernRuntime/N64Recomp/include")
target_compile_definitions(ultramodern PRIVATE RECOMP_FAIR_GFX_QUEUE)
file(GLOB game_c "${generated}/*.c")
file(GLOB game_cpp "${generated}/*.cpp")
add_library(probe_game STATIC ${game_c} ${game_cpp})
target_include_directories(probe_game PRIVATE "${repo}/include" "${repo}/lib/N64ModernRuntime/N64Recomp/include")
target_compile_options(probe_game PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wno-implicit-function-declaration>)
add_executable(dk64_runtime_probe main.cpp
    ../vita/game_support.cpp ../vita/register_overlays.cpp ../vita/timing.cpp
    "${repo}/src/game/rom_decompression.cpp" "${repo}/rsp/n_aspMain.cpp")
target_include_directories(dk64_runtime_probe PRIVATE "${repo}/include" "${generated}"
    "${repo}/lib/N64ModernRuntime/thirdparty/sse2neon")
target_link_libraries(dk64_runtime_probe PRIVATE probe_game librecomp ultramodern rt64 pthread)
target_link_options(dk64_runtime_probe PRIVATE -Wl,--gc-sections -Wl,--wrap=osPiRawStartDma_recomp)
add_executable(dk64_timing_checks ../vita/timing.cpp ../vita/test_timing.cpp)
target_include_directories(dk64_timing_checks PRIVATE "${repo}/lib/N64ModernRuntime/N64Recomp/include")
add_test(NAME dk64_timing_checks COMMAND dk64_timing_checks)
if(PROBE_GL)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
    target_sources(dk64_runtime_probe PRIVATE egl.cpp)
    target_link_libraries(dk64_runtime_probe PRIVATE PkgConfig::EGL)
    target_compile_definitions(dk64_runtime_probe PRIVATE PROBE_GL)
    add_executable(rt64_gl_checks gl_checks.cpp egl.cpp)
    target_link_libraries(rt64_gl_checks PRIVATE rt64 PkgConfig::EGL)
    add_test(NAME rt64_gl_checks COMMAND rt64_gl_checks)
    set_tests_properties(rt64_gl_checks PROPERTIES ENVIRONMENT "GALLIUM_DRIVER=softpipe")
endif()
